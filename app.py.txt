st.write("Completa el formulario para unirte al nuevo eón.")

# FORMULARIO
with st.form("formulario_registro", clear_on_submit=True):
    nombre = st.text_input("Nombre completo")
    email = st.text_input("Correo electrónico")
    celular = st.text_input("Número de celular")

    enviado = st.form_submit_button("Suscribirse")

    if enviado:
        # VALIDACIÓN BÁSICA
        if not nombre or not email or not celular:
            st.error("Todos los campos son obligatorios. Completa el formulario.")
        elif "@" not in email:
            st.error("Ingresa un correo electrónico válido.")
        else:
            try:
                # GUARDAR EN SUPABASE
                data = {
                    "nombre": nombre,
                    "email": email,
                    "celular": celular
                }
                supabase.table("suscriptores").insert(data).execute()

                # MENSAJE DE CONFIRMACIÓN
                st.success(f"¡{nombre}, tu registro ha sido exitoso!")
                st.balloons()
                st.write("Bienvenido a EONIA. El nuevo eón comienza contigo.")

            except Exception as e:
                error_msg = str(e)
                if "duplicate key" in error_msg.lower():
                    st.error("Este correo electrónico ya está registrado.")
                else:
                    st.error(f"Error al guardar: {error_msg}")

# MOSTRAR REGISTROS (OPCIONAL, SOLO PARA VERIFICAR)
if st.checkbox("Ver suscriptores registrados"):
    try:
        response = supabase.table("suscriptores").select("*").execute()
        if response.data:
            st.dataframe(response.data)
        else:
            st.info("No hay suscriptores aún.")
    except:
        st.warning("No se pudieron cargar los registros.")
